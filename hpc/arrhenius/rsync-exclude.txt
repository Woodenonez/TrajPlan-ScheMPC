/.venv/
/mpc_solver/

# Editor and OS state.
/.idea/
.DS_Store

# Editable installs. These record absolute paths from the machine that built

*.egg-info/

__pycache__/
*.pyc
.pytest_cache/
.mypy_cache/

# Results flow cluster -> laptop only ("pull"), never the other way: push has no --update, so
